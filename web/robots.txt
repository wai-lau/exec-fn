# wai-lau.net
#
# Almost every page here sits behind a login or a Turnstile challenge, so a
# crawler that follows those links collects nothing but redirects to the two
# login forms. The landing page and the résumé are the parts meant to be
# found; everything else is listed here so a crawler doesn't spend its budget
# discovering that for itself.
#
# AI/training crawlers get the opposite default, in their own group at the
# bottom: everything blocked except the résumé.

User-agent: *

# Full-auth pages (session cookie).
Disallow: /rd
Disallow: /hq
Disallow: /debug
Disallow: /emet

# Guest-gated pages (Cloudflare Turnstile).
Disallow: /UI
Disallow: /graph
Disallow: /hosaka
Disallow: /mtg
Disallow: /nightfall
Disallow: /printer
Disallow: /security
Disallow: /tarot

# These rules are PREFIXES, not paths, so "/printer" also covers the landing
# page's own printer.png icon. A renderer fetches that, so allow it back --
# longest match wins, and Allow wins a tie (RFC 9309).
Allow: /printer.png

# Login forms: nothing to index, and an indexed login only draws credential
# noise to it.
Disallow: /login
Disallow: /guest
Disallow: /guest-login

# Machine surfaces -- JSON APIs, websockets, the data dir, the game bundle.
Disallow: /api/
Disallow: /ws/
Disallow: /data/
Disallow: /nightfall-game/

# ── AI / training crawlers ───────────────────────────────────────────────
# A crawler obeys exactly ONE group -- the most specific User-agent match --
# so this block REPLACES the rules above rather than adding to them, and has
# to restate its own. Everything is closed; the résumé and its PDF are the
# one thing published to be read, and "/recruiter" being a prefix covers the
# page's stylesheet and script with it.
#
# Google-Extended and Applebot-Extended are not crawlers -- they are opt-out
# tokens for Gemini and Apple Intelligence training. Googlebot and Applebot
# still index under the group above, so search is unaffected.
